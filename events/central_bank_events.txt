# MTTH informational event with explicit cooldown guard.
country_event = {
    id = central_bank.900
    title = central_bank.900.t
    desc = central_bank.900.d
    picture = GFX_report_event_default
    mean_time_to_happen = {
        days = 120
    }
    trigger = {
        central_bank_is_initialized = yes
        NOT = { has_country_flag = central_bank_info_cd }
    }

    immediate = {
        set_timed_country_flag = { flag = central_bank_info_cd days = 120 }
    }

    option = {
        name = central_bank.900.a
    }
}
