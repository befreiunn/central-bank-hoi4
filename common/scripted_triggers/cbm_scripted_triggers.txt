}

cbm_is_debt_critical_trigger = {
    check_variable = { cbm_debt_to_gdp > 120 }
}

cbm_is_fx_crisis_trigger = {
    check_variable = { cbm_fx_pressure > 75 }
    check_variable = { cbm_fx_reserves < 25 }
}

    check_variable = { cbm_policy_rate > 1 }
    check_variable = { cbm_financial_confidence > 30 }
}

    check_variable = { cbm_crisis_risk > 80 }
}

cbm_is_low_confidence_trigger = {
    check_variable = { cbm_financial_confidence < 35 }
}

cbm_is_banking_fragile_trigger = {

}
